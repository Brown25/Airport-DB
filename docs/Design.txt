-- Employee Table
CREATE TABLE employee (
    ssn CHAR(9) PRIMARY KEY,
    name TEXT NOT NULL,
    password TEXT NOT NULL,
    address TEXT,
    phone TEXT,
    salary NUMERIC(100, 2) CHECK (salary >= 0)
);

-- Manager Table
CREATE TABLE manager (
    ssn CHAR(9) PRIMARY KEY,
    FOREIGN KEY (ssn) REFERENCES employee(ssn) ON DELETE CASCADE
);

-- Technician Table
CREATE TABLE technician (
    ssn CHAR(9) PRIMARY KEY,
    FOREIGN KEY (ssn) REFERENCES employee(ssn) ON DELETE CASCADE
);

-- ATC (Air Traffic Controller) Table
CREATE TABLE atc (
    ssn CHAR(9) PRIMARY KEY,
    FOREIGN KEY (ssn) REFERENCES employee(ssn) ON DELETE CASCADE
);

-- Airplane Model Table
CREATE TABLE airplane_model (
    model_number TEXT PRIMARY KEY,
    capacity INTEGER NOT NULL CHECK (capacity >= 0),
    weight INTEGER NOT NULL CHECK (weight >= 0)
);

-- Airplane Table
CREATE TABLE airplane (
    reg_number TEXT PRIMARY KEY,
    model_number TEXT NOT NULL,
    FOREIGN KEY (model_number) REFERENCES airplane_model(model_number) ON DELETE CASCADE
);

-- Expert Table (Multi-Column Primary Key!)
CREATE TABLE expert (
    ssn CHAR(9) NOT NULL,
    model_number TEXT NOT NULL,
    PRIMARY KEY (ssn, model_number),
    FOREIGN KEY (ssn) REFERENCES employee(ssn) ON DELETE CASCADE,
    FOREIGN KEY (model_number) REFERENCES airplane_model(model_number) ON DELETE CASCADE
);

-- FAA Test Table
CREATE TABLE faa_test (
    test_number INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    max_score INTEGER NOT NULL CHECK (max_score > 0)
);

-- Test Event Table (Multiple Foreign Keys!)
CREATE TABLE test_event (
    test_number INTEGER NOT NULL,
    ssn CHAR(9) NOT NULL,
    reg_number TEXT NOT NULL,
    date DATE NOT NULL,
    duration INTERVAL,
    score INTEGER NOT NULL CHECK (score >= 0),

    PRIMARY KEY (test_number, ssn, reg_number, date),

    FOREIGN KEY (test_number) REFERENCES faa_test(test_number) ON DELETE CASCADE,
    FOREIGN KEY (ssn) REFERENCES employee(ssn) ON DELETE CASCADE,
    FOREIGN KEY (reg_number) REFERENCES airplane(reg_number) ON DELETE CASCADE
);



	•	ON DELETE CASCADE means if an employee, airplane, or test is deleted, related rows also delete automatically (can be changed if you want).
	•	CHECK constraints make sure salary, capacity, weight, and scores are reasonable (positive or non-negative numbers).
	•	test_event has a 4-column composite primary key (test_number + ssn + reg_number + date) because it’s a log of multiple related entities.
	•	Passwords should be stored hashed in a real system, not plain text.
